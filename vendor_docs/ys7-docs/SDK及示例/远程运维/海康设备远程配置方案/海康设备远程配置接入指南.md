# 海康设备远程配置说明

## 配置步骤

- 下载插件并安装运行

  远程配置插件：[EzvizWebControlSign.exe](https://download2.ys7.com/openweb/web/EzvizWebControlSign.exe)
- 下载demo示例

  该示例为原生jquery写法，支持运行查看页面，可根据需要自行改造，应用到自己项目中 [demo示例](https://fastgw.ys7.com/group35/M00/2B/5A/CtxID2N81NSAV0DGAAHiaeJJlqk630.zip)
- react项目实际应用

  1.以开放平台设备列表的在线配置功能为例，**首先从上述demo示例中获取js文件**：jsWebControl-1.0.0.min.js复制到本地项目中，可通过  `<script >`标签的形式引入到项目中，然后就可在相应页面进行开发示例代码如下（代码中各字段都添加了注释说明）：  
  2.字段说明:

  | 字段名 | 描述 |
  | --- | --- |
  | accessToken | 账户标识 |
  | appkey | 账户标识 |
  | expireTime | 过期时间 |
  | subSerial | 设备序列号 |
  | devTag | 设备类型，支持多种设备类型：ipc，nvr，dvr |

  3.注意事项

  并不是所有设备都支持远程在线配置，**目前该方案只支持设备版本在 v4.20.xxx以上的设备**

```
// FileName: 点击在线配置后执行的方法
handleOnlineConfig(record) {
    const that = this;
    this.setState({ configLoading: true, id: record.subSerial })  //在线配置启动较慢，添加loading状态，同时传入需要配置的设备序列号
    const { accessToken, appKey, expireTime } = this.state;  //拿到token,appKey和过期时间

    let oWebControl = null;
    let m_uuid = this.uuid();// 每次都需要随机生成的uuid
    let m_delegateState = 0;// 固定值
    let iServicePort = 14490;   // 服务端口
    oWebControl = new WebControl({      //手动设置Token，会自动连接Web控件
      szPluginContainer: "playWnd",  // 固定值
      iServicePortStart: iServicePort, // 对应 LocalServiceConfig.xml 中的ServicePortStart值
      iServicePortEnd: iServicePort + 9, // 对应 LocalServiceConfig.xml 中的ServicePortEnd值
      szClassId: "23BF3B0A-2C56-4D97-9C03-0CB103AA8F11",   // 固定值
      cbConnectSuccess: function () {
        that.setState({ configLoading: false })    //配置页面成功打开后关闭loading
        oWebControl.JS_SetCloudCallback({
          cbCloudCallback: function (oData) {
          }
        });
        oWebControl.JS_CloudRequest({
          funcName: "setToken",
          arguments: {
            uuid: m_uuid,   //随机生成
            delegateState: m_delegateState,  //固定值
            accessToken: accessToken,
            areaDomain: 'https://open.ys7.com' ,   //域名，根据你的项目实际域名填写
            expireTime: expireTime,
            appKey: appKey,
            juridicalPersonFlag: true,// 固定值
            challenge: "",// 固定值
            iterateNum: 100,// 固定值
            sessionId: "",// 固定值
            user: ""// 固定值
          }
        });
        const auth = window.btoa('');
        const lan = 'zh';
        var port = oWebControl.JS_GetConnectPort();
       //拼接配置页面的地址，并打开
        window.open("http://127.0.0.1:" + port + "/dispatch.html?serial=" + record.subSerial + "&type=" + (record.devTag).toLowerCase() + "&auth=" + auth + "&lan=" + lan + "&uuid=" + m_uuid + "&state=0")
      },
     //启动失败的回调，启动时会自动检测插件是否安装，如果未安装，可以在弹窗此提醒用户下载并安装
      cbConnectError: function () {
        that.setState({ configLoading: false })
        confirm({
          title: <div style={{ fontSize: 16, color: '#262626', lineHeight: '24px', fontWeight: 500 }}>您好，请先确认已安装并运行“海康设备远程配置插件”，如果还未安装插件，请先下载安装。</div>,
          icon: <Icon type="exclamation-circle" theme="filled" style={{ color: "#FAAD14" }} />,
          content: '',
          okText: "下载插件",
          cancelText: '关闭',
          centered: true,
          className: 'onlineModal',
          zIndex: 99999,
          onOk() {
            window.location.href = **这里填写插件的下载地址**
          },
          onCancel() {
            console.log('Cancel');
          },
        });
      }
    });
  }
```
